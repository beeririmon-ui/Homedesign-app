  return c.body.order_id!;
}

/** a network-like provider: slow, counts every invoice it issues; options: can be asked (find), dedupes by key */
function counting(o: { delayMs?: number; find?: boolean; idempotent?: boolean; hangUntil?: Promise<void> } = {}) {
  const issued: string[] = [];
  const byKey = new Map<string, IssuedInvoice>();
  const p: InvoiceProvider & { issued: string[] } = {
    id: 'qa-counting',
    idempotent: !!o.idempotent,
    issued,
    async issue(input, { idempotency_key }) {
      if (o.hangUntil) await o.hangUntil;
      else await new Promise((r) => setTimeout(r, o.delayMs ?? 20));
      if (o.idempotent && byKey.has(idempotency_key)) return byKey.get(idempotency_key)!;
      issued.push(`${input.order_id}#${issued.length + 1}`);
      const inv = { invoice_id: `QA-INV-${input.order_id}-${issued.length}`, url: null };
      byKey.set(idempotency_key, inv);
      return inv;
    },
  };
  if (o.find) p.find = async (id) => byKey.get(id) ?? null;
  return p;
}
/** the Worker dies after the provider answered and before anything is written: the next batch() throws, once */
function dieAtNextBatch(e: Env): Env {
  let armed = true;
  const db = e.DB as unknown as { batch: (s: unknown[]) => Promise<unknown> };
  const real = db.batch.bind(db);
  return {
    ...e,
    DB: new Proxy(e.DB, {
      get(t, k) {
        if (k === 'batch')
          return async (s: unknown[]) => {
            if (armed) {
              armed = false;
              throw new Error('qa: the isolate died');
            }
            return real(s);
          };
        const v = (t as unknown as Record<string | symbol, unknown>)[k];
        return typeof v === 'function' ? (v as (...a: unknown[]) => unknown).bind(t) : v;
      },
    }),
  };
}
const invoicesOf = (id: string) => count("SELECT COUNT(*) AS n FROM audit_log WHERE action = 'order.invoiced' AND subject = ?", id);

describe('QA ת16: one invoice per order (real seed)', () => {
  it('5 runs at once, a slow provider that dedupes nothing and cannot be asked: one invoice', async () => {
    const id = await paid();
    const p = counting({ delayMs: 40 });
    const out = await Promise.all(Array.from({ length: 5 }, () => processInvoiceJob(env, id, p)));
    expect(p.issued).toHaveLength(1);
    expect(out.filter((o) => o === 'invoiced')).toHaveLength(1);
    expect(out.filter((o) => o === 'retry')).toHaveLength(4);
    expect(await invoicesOf(id)).toBe(1);
    // later runs (the retries) skip
    expect(await Promise.all([processInvoiceJob(env, id, p), processInvoiceJob(env, id, p)])).toEqual(['skipped', 'skipped']);
    expect(p.issued).toHaveLength(1);
  });

  it('a crash after the provider issued, before anything was written: retries within the lease wait; after it a provider that can be asked records the first invoice', async () => {
    const id = await paid();
    const p = counting({ find: true });
    let t = Date.now();
    const now = () => t;
    await expect(processInvoiceJob(dieAtNextBatch(env), id, p, { now })).rejects.toThrow('isolate died');
    expect(p.issued).toHaveLength(1);
    expect((await one<{ state: string }>('SELECT state FROM invoices WHERE order_id = ?', id)).state).toBe('claimed');
    t += 60_000;
    expect(await processInvoiceJob(env, id, p, { now })).toBe('retry');
    t += INVOICE_LEASE_MS;
    expect(await processInvoiceJob(env, id, p, { now })).toBe('invoiced');
    expect(p.issued).toHaveLength(1);
    expect((await one<{ invoice_number: string }>('SELECT invoice_number FROM orders WHERE id = ?', id)).invoice_number).toBe(`QA-INV-${id}-1`);
    expect(await invoicesOf(id)).toBe(1);
  });

  it('the same crash with a provider that dedupes by key (cannot be asked): the retry sends the same key and gets the first invoice', async () => {
    const id = await paid();
    const p = counting({ idempotent: true });
    let t = Date.now();
    const now = () => t;
    await expect(processInvoiceJob(dieAtNextBatch(env), id, p, { now })).rejects.toThrow();
    t += INVOICE_LEASE_MS + 1;
    expect(await processInvoiceJob(env, id, p, { now })).toBe('invoiced');
    expect(p.issued).toHaveLength(1);
    expect((await one<{ invoice_number: string }>('SELECT invoice_number FROM orders WHERE id = ?', id)).invoice_number).toBe(`QA-INV-${id}-1`);
  });

  it('the same crash with a provider that can neither be asked nor dedupe: unresolved, and it stays so (runs, an hour later, the queue) for a person', async () => {
    const id = await paid();
    const p = counting({});
    let t = Date.now();
    const now = () => t;
    const err = vi.spyOn(console, 'error').mockImplementation(() => undefined);
    await expect(processInvoiceJob(dieAtNextBatch(env), id, p, { now })).rejects.toThrow();
    t += INVOICE_LEASE_MS + 1;
    expect(await processInvoiceJob(env, id, p, { now })).toBe('unresolved');
    for (const dt of [1, 60_000, 3_600_000]) {
      t += dt;
      expect(await processInvoiceJob(env, id, p, { now })).toBe('unresolved');
    }
    expect(p.issued).toHaveLength(1);
    expect(await count("SELECT COUNT(*) AS n FROM audit_log WHERE action = 'invoice.unresolved' AND subject = ?", id)).toBe(1);
    expect((await one<{ invoice_number: string | null }>('SELECT invoice_number FROM orders WHERE id = ?', id)).invoice_number).toBeNull();
    // through the queue (the configured provider, the mock): the message is acknowledged, never retried, nothing issued
    const ack = vi.fn();
    const retry = vi.fn();
    await worker.queue({ messages: [{ body: { kind: 'issue-invoice', order_id: id }, attempts: 1, ack, retry }] } as unknown as MessageBatch<SupplierJob>, env);
    expect(ack).toHaveBeenCalledTimes(1);
    expect(retry).not.toHaveBeenCalled();
    expect((await one<{ state: string }>('SELECT state FROM invoices WHERE order_id = ?', id)).state).toBe('unresolved');
    expect((await one<{ invoice_number: string | null }>('SELECT invoice_number FROM orders WHERE id = ?', id)).invoice_number).toBeNull();
    expect(err).toHaveBeenCalled();
    err.mockRestore();
  });

  it('adversarial: a provider call longer than the lease (a hung request); the second run takes the claim over while the first is still waiting', async () => {
    const id = await paid();
    let release!: () => void;
    const hang = new Promise<void>((r) => (release = r));
    const slow = counting({ find: true, hangUntil: hang }); // can be asked, does not dedupe
    let t = Date.now();
    const now = () => t;
    const first = processInvoiceJob(env, id, slow, { now });
    await new Promise((r) => setTimeout(r, 20));
    t += INVOICE_LEASE_MS + 1;
    // a provider that cannot find the invoice that is still being issued: the takeover issues
    const fast = { ...counting({ find: true }), id: 'qa-counting' } as InvoiceProvider & { issued: string[] };
    const second = await processInvoiceJob(env, id, fast, { now });
    release();
    const a = await first;
    // what happened, recorded for the report (not asserted as a pass or a fail)
    const recorded = await one<{ provider_invoice_id: string }>('SELECT provider_invoice_id FROM invoices WHERE order_id = ?', id);
    console.log(`QA lease overrun: first=${a} second=${second} issued=${slow.issued.length + fast.issued.length} recorded=${recorded.provider_invoice_id}`);
    expect(slow.issued.length + fast.issued.length).toBeGreaterThanOrEqual(1);
  });
});

describe('QA ת17: production fails closed on every provider variable (real seed)', () => {
  const PROD_ENVS = ['production', undefined, '', 'Production', 'prod', 'PRODUCTION ', 'staging'];
  const VALUES: Record<'PAYMENT_PROVIDER' | 'INVOICE_PROVIDER' | 'SUPPLIER_PROVIDER', { bad: (string | undefined)[]; good: string[] }> = {
    PAYMENT_PROVIDER: { bad: [undefined, '', 'mock', 'Mock', 'MOCK', ' mock', 'Grow', 'GROW', 'grow ', 'UNDECIDED', 'stripe', 'tranzila'], good: ['grow'] },
    INVOICE_PROVIDER: { bad: [undefined, '', 'mock', 'Mock', 'Morning', 'GROW', 'morning ', 'UNDECIDED', 'icount'], good: ['grow', 'morning'] },
    SUPPLIER_PROVIDER: { bad: [undefined, '', 'mock', 'MOCK', 'CJ', 'Cj', ' cj', 'cj ', 'cjdropshipping', 'UNDECIDED', 'zendrop'], good: ['cj'] },
  };
  const prodEnv = (E: string | undefined, over: Partial<Env>): Env => ({ ...env, ENVIRONMENT: E as Env['ENVIRONMENT'], ...over });

  it('providerOf: in every production spelling, only the exact provider is a provider; mock never', () => {
    const rows: string[] = [];
    for (const E of PROD_ENVS)
      for (const [name, { bad, good }] of Object.entries(VALUES)) {
        for (const v of bad) {
          const c = providerOf({ ENVIRONMENT: E, [name]: v }, name as keyof typeof VALUES);
          if (c.ok) rows.push(`${E}/${name}=${JSON.stringify(v)} ran ${c.id}`);
        }
        for (const v of good) {
          const c = providerOf({ ENVIRONMENT: E, [name]: v }, name as keyof typeof VALUES);
          if (!c.ok || c.mock) rows.push(`${E}/${name}=${v} refused`);
        }
      }
    expect(rows).toEqual([]);
  });

  it('the deploy check flags every bad value in a production block (UNDECIDED is the one placeholder it lets through)', () => {
    const missed: string[] = [];
    for (const [name, { bad }] of Object.entries(VALUES))
      for (const v of bad) {
        if (v === 'UNDECIDED') continue;
        const vars = { ENVIRONMENT: 'production', PAYMENT_PROVIDER: 'UNDECIDED', INVOICE_PROVIDER: 'UNDECIDED', SUPPLIER_PROVIDER: 'cj', [name]: v };
        if (v === undefined) delete (vars as Record<string, unknown>)[name];
        if (!configIssues(vars).length) missed.push(`${name}=${JSON.stringify(v)}`);
      }
    expect(missed).toEqual([]);
    expect(configIssues({ PAYMENT_PROVIDER: 'mock', INVOICE_PROVIDER: 'mock', SUPPLIER_PROVIDER: 'mock' }).length).toBeGreaterThan(0);
  });

  it('the payment: every bad value, and grow (no documentation), in every production spelling: 5xx and no order left behind', async () => {
    const c = await cart([{ variant_id: PLAIN }]);
    const bad: string[] = [];
    for (const E of PROD_ENVS)
      for (const v of [...VALUES.PAYMENT_PROVIDER.bad, 'grow']) {
        const e = prodEnv(E, { PAYMENT_PROVIDER: v as string });
        const before = await count('SELECT COUNT(*) AS n FROM orders');
        const r = await req('/api/checkout', json(buyer(c)), e);
        const after = await count('SELECT COUNT(*) AS n FROM orders');
        if (r.status < 500 || after !== before) bad.push(`${E}/${JSON.stringify(v)}: ${r.status}, orders ${before}→${after}`);
        // the mock's pages and webhook: closed
        const m = await req('/mock-pay/anything', undefined, e);
        if (m.status !== 404) bad.push(`${E}/${JSON.stringify(v)}: /mock-pay ${m.status}`);
        const w = await req('/api/webhooks/payment/mock', { method: 'POST', body: '{}', headers: { 'content-type': 'application/json' } }, e);
        // 401 would mean the mock's verifier ran (the mock is live); 200 that an event was taken
        if (w.status === 200 || w.status === 401) bad.push(`${E}/${JSON.stringify(v)}: mock webhook ${w.status}`);
      }
    expect(bad).toEqual([]);
  });

  it('the supplier job: every bad value in every production spelling leaves a paid order paid (supplier_error at attempt 5), never sent; cj without a key: the same', async () => {
    const bad: string[] = [];
    const err = vi.spyOn(console, 'error').mockImplementation(() => undefined);
    for (const E of PROD_ENVS.slice(0, 3))
      for (const v of [...VALUES.SUPPLIER_PROVIDER.bad, 'cj']) {
        const id = await paid();
        const e = prodEnv(E, { SUPPLIER_PROVIDER: v as string });
        for (let attempt = 1; attempt <= 8; attempt++) {
          const r = await processSupplierJob(e, { kind: 'submit-order', order_id: id }, attempt);

        }

        if (sent) bad.push(`${E}/${JSON.stringify(v)}: a shipment marked submitted`);
      }

  });

  it('the invoice job: every bad value, and the real ids (not built), in every production spelling: retried, nothing recorded', async () => {

describe('QA the Mediterranean print: its real card has no named set, and a set without its parts is refused', () => {
  it('the card (8 prints for a set of 3): setComposition is incomplete at every size; a set variant carrying that is 422 set_unavailable, and a cart that held it loses it before payment', async () => {
    const { setComposition } = await import('../../tools/lib/lines');
    const { readdirSync, statSync } = await import('node:fs');
    const find = (d: string): string | null => {
      for (const n of readdirSync(d)) {
        const p = resolve(d, n);
        if (statSync(p).isDirectory()) { const r = find(p); if (r) return r; } else if (n === 'framed-art-aliexpress-mediterranean-wabi-sabi-poster-802890.json') return p;
      }
      return null;
    };
    const card = JSON.parse(readFileSync(find(resolve(here, '../../../data/products'))!, 'utf8')) as { options: { axes: { id: string; values: { id: string }[] }[]; skus: { id: string; values: Record<string, string> }[]; default_sku: string } };
    const story = JSON.parse(readFileSync(resolve(here, '../../../data/story/story.json'), 'utf8')) as { set_axes: Record<string, { axis: string; note_he: string; parts?: string[] }> };
    const set = story.set_axes['framed-art-aliexpress-mediterranean-wabi-sabi-poster-802890']!;
    const axis = card.options.axes.find((a) => a.id === set.axis);
    const reasons = new Set<string>();
    for (const sku of card.options.skus.filter((k) => k.values[set.axis] === card.options.skus.find((d) => d.id === card.options.default_sku)!.values[set.axis])) {
      const c = setComposition(set, axis, card.options.skus, sku, 3);
      expect(c.skus, sku.id).toBeNull();
      reasons.add(c.incomplete!);
    }
    // the server's side of it, with that very reason, on the triptych's set variant (the D1 row the build would write)
    const v = trip.variants_full!.find((x) => x.set_skus)!;
    const { id } = (await (await req('/api/cart', { method: 'POST' })).json()) as { id: string };
    expect((await req(`/api/cart/${id}/items`, json({ variant_id: v.id, qty: 1 }, 'PUT'))).status).toBe(200);
    await env.DB.prepare('UPDATE product_variants SET set_incomplete = ?, set_skus_json = NULL WHERE id = ?').bind([...reasons][0], v.id).run();
    const again = await req(`/api/cart/${id}/items`, json({ variant_id: v.id, qty: 2 }, 'PUT'));
    expect(again.status).toBe(422);
    expect(((await again.json()) as { error: string }).error).toBe('set_unavailable');
    const view = (await (await req(`/api/cart/${id}`)).json()) as { lines: unknown[]; removed?: number };
    expect(view.lines).toHaveLength(0);
    expect(view.removed).toBe(1);
    const co = await req('/api/checkout', json(buyer(id)));
    expect([409]).toContain(co.status);
    console.log(`QA Mediterranean: ${[...reasons].join(' | ')}; checkout after: ${co.status} ${((await co.json()) as { error: string }).error}`);
  });
});
