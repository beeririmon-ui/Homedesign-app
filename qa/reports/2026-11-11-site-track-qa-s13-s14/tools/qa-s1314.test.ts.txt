/** a payment provider like a network one: counts every refund call and every refund it makes; dedupes and can be asked on demand */
function payments(o: { idempotent?: boolean; find?: boolean; delayMs?: number } = {}) {
  const calls: string[] = [];
  const made: { key: string; amount: number }[] = [];
  const byKey = new Map<string, IssuedRefund>();
  const p: PaymentProvider & { calls: string[]; made: typeof made } = {
    id: 'mock',
    calls,
    made,
    refundsIdempotent: o.idempotent ?? true,
    async createSession() {
      throw new Error('not used');
    },
    async verifyWebhook() {
      return { ok: false, reason: 'not used' };
    },
    async refund(input: RefundInput, { idempotency_key }) {
      calls.push(idempotency_key);
      await new Promise((r) => setTimeout(r, o.delayMs ?? 15));
      if ((o.idempotent ?? true) && byKey.has(idempotency_key)) return byKey.get(idempotency_key)!;
      if (input.refunded_before_agorot + input.amount_agorot > input.paid_agorot) throw new RefundRefused('exceeds');
      made.push({ key: idempotency_key, amount: input.amount_agorot });
      const r = { refund_id: `QA-RF-${made.length}` };
      byKey.set(idempotency_key, r);
      return r;
    },
  };
  if (o.find) p.findRefund = async (k) => byKey.get(k) ?? null;
  return p;
}

/** an invoice provider that counts invoices and credit notes; can hang, can be asked, can echo the buyer in its error */
function invoicing(o: { find?: boolean; idempotent?: boolean; hang?: Promise<void>; failWith?: (input: { customer?: { full_name: string } }) => Error } = {}) {
  const issued: string[] = [];
  const credits: { key: string; amount: number }[] = [];
  const byKey = new Map<string, { invoice_id: string; url: string | null }>();
  const cnByKey = new Map<string, { credit_note_id: string; url: string | null }>();
  const p = {
    id: 'qa-inv',
    idempotent: !!o.idempotent,
    issued,
    credits,
    async issue(input: { order_id: string; customer: { full_name: string } }, { idempotency_key }: { idempotency_key: string }) {
      if (o.hang) await o.hang;
      if (o.failWith) throw o.failWith(input);
      if (o.idempotent && byKey.has(idempotency_key)) return byKey.get(idempotency_key)!;
      issued.push(input.order_id);
      const inv = { invoice_id: `QA-INV-${issued.length}`, url: null };
      byKey.set(idempotency_key, inv);
      return inv;
    },
    async creditNote(input: { amount_agorot: number }, { idempotency_key }: { idempotency_key: string }) {
      if (o.hang) await o.hang;
      if (o.idempotent && cnByKey.has(idempotency_key)) return cnByKey.get(idempotency_key)!;
      credits.push({ key: idempotency_key, amount: input.amount_agorot });
      const cn = { credit_note_id: `QA-CN-${credits.length}`, url: null };
      cnByKey.set(idempotency_key, cn);
      return cn;
    },
    ...(o.find
      ? {
          find: async (id: string) => byKey.get(id) ?? null,
          findCreditNote: async (k: string) => cnByKey.get(k) ?? null,
        }
      : {}),
  };
  return p as unknown as InvoiceProvider & { issued: string[]; credits: typeof credits };
}

/** a supplier that records every submit and every cancellation request; can hang past the deadline and still "create" */
function supplier(o: { hangMs?: number; cancel?: 'canceled' | 'requested' | 'refused'; failWith?: (i: SupplierOrderInput) => Error } = {}) {
  const orders: string[] = [];
  const cancels: string[] = [];
  const c: SupplierClient & { orders: string[]; cancels: string[] } = {
    id: 'mock',
    orders,
    cancels,
    async submitOrder(input) {
      if (o.failWith) throw o.failWith(input);
      orders.push(input.order_id); // created at the supplier the moment it is received
      if (o.hangMs) await new Promise((r) => setTimeout(r, o.hangMs));
      return { supplier_order_id: `QA-SUP-${input.order_id}`, raw: {} };
    },
    async cancelOrder(sid) {
      cancels.push(sid);
      return { outcome: o.cancel ?? 'requested' };
    },
    async tracking() {
      return null;
    },
  };
  return c;
}

const b64u = (b: ArrayBuffer | Uint8Array | string) =>
  Buffer.from(typeof b === 'string' ? b : b instanceof Uint8Array ? b : new Uint8Array(b))
    .toString('base64')
    .replace(/=+$/, '')
    .replace(/\+/g, '-')
    .replace(/\//g, '_');
async function rsaKey(kid: string) {
  const k = (await crypto.subtle.generateKey({ name: 'RSASSA-PKCS1-v1_5', modulusLength: 2048, publicExponent: new Uint8Array([1, 0, 1]), hash: 'SHA-256' }, true, [
    'sign',
    'verify',
  ])) as CryptoKeyPair;
  const jwk = { ...(await crypto.subtle.exportKey('jwk', k.publicKey)), kid, alg: 'RS256', use: 'sig' };
  return { k, jwk };
}
async function jwt(key: CryptoKey | null, header: object, payload: object) {
  const h = b64u(JSON.stringify(header));
  const p = b64u(JSON.stringify(payload));
  const sig = key ? b64u(await crypto.subtle.sign('RSASSA-PKCS1-v1_5', key, new TextEncoder().encode(`${h}.${p}`))) : '';
  return `${h}.${p}.${sig}`;
}

describe('QA H (ת27): every sold variant carries a SKU of its card', () => {
  type Card = { id?: string; supplier?: { name?: string; sku?: string; product_url?: string } | null; options?: { default_sku?: string; skus?: { id: string }[] } | null };
  const card = (p: string) => JSON.parse(readFileSync(resolve(REPO, p), 'utf8')) as Card;
  /** my own reading of the cached CJ product (data/sources/cj/<pid>.json): its product-level code and its variants' SKUs */
  const cjOf = (c: Card): { product: string | null; variants: string[] } | null => {
    if (!/^CJ/i.test(c.supplier?.name ?? '')) return null;
    const pid = /-p-(\d+)\.html/.exec(c.supplier?.product_url ?? '')?.[1];
    const f = pid ? resolve(REPO, 'data/sources/cj', `${pid}.json`) : null;
    if (!f || !existsSync(f)) return { product: null, variants: [] };
    const pl = (JSON.parse(readFileSync(f, 'utf8')) as { sections?: { product?: { payload?: { productSku?: string; variants?: { variantSku?: string }[] } } } }).sections?.product?.payload;
    return { product: pl?.productSku ?? null, variants: (pl?.variants ?? []).map((v) => v.variantSku!).filter(Boolean) };
  };
  const excl = JSON.parse(readFileSync(resolve(REPO, 'data/catalog/sale-exclusions.json'), 'utf8')) as { exclusions: { card_id: string }[] };
  const exclCards = new Set(excl.exclusions.map((e) => e.card_id));

  it('both catalogs: each sold variant → its card; with options: one of options.skus; without: the CJ product\'s variant, never its product code', () => {
    const issues: string[] = [];
    const noOpt: string[] = [];
    let checked = 0;
    for (const [name, cat] of [
      ['shop', shop],
      ['story', story],
      ['test-page', testPage],
    ] as const)
      for (const p of cat.products) {
        if (!p.source_path || !existsSync(resolve(REPO, p.source_path))) continue;
        const c = card(p.source_path);
        const opts = new Set((c.options?.skus ?? []).map((s) => s.id));
        const cj = cjOf(c);
        for (const v of p.variants_full ?? []) {
          if (!v.active || v.set_part || (v as { sale_excluded?: string | null }).sale_excluded) continue;
          checked++;
          const skus = v.set_skus?.length ? v.set_skus : [v.supplier_sku];
          for (const s of skus) {
            if (!s) issues.push(`${name} ${v.id}: no SKU`);
            else if (opts.size) {
              if (!opts.has(s)) issues.push(`${name} ${v.id}: ${s} not in the card's ${opts.size} option SKUs`);
            } else if (cj) {
              noOpt.push(`${name} ${p.id}: card ${c.supplier?.sku} → sold ${s} (cache: product ${cj.product}, ${cj.variants.length} variant${cj.variants.length === 1 ? '' : 's'}${cj.variants.length <= 3 ? ` ${cj.variants.join(',')}` : ''})`);
              if (cj.product && s === cj.product) issues.push(`${name} ${v.id}: ${s} is the CJ product-level code`);
              if (cj.variants.length && !cj.variants.includes(s)) issues.push(`${name} ${v.id}: ${s} is not one of the CJ product's variants`);
            }
          }
        }
      }
    console.log(`QA H1: ${checked} sold variants checked (both catalogs); issues ${issues.length}${issues.length ? `\n  ${issues.slice(0, 20).join('\n  ')}` : ''}\n  CJ cards without options:\n  ${[...new Set(noOpt)].join('\n  ')}`);
    expect(issues).toEqual([]);
  });

  it('one paid order with every product that has sale exclusions or a single SKU: the supplier request carries option SKUs', async () => {
    const seen = new Set<string>();
    const picks = [...shop.products, ...story.products].filter((p) => {
      if (seen.has(p.id)) return false;
      seen.add(p.id);
      if (!p.price_agorot || p.availability.status !== 'available' || p.pack) return false;
      const c = card(p.source_path);
      return exclCards.has(c.id ?? '') || (c.options?.skus ?? []).length <= 1;
    });
    const items = picks.map((p) => ({ variant_id: p.variants[0]!.id }));
    const o = await paid(items);
    expect(await processSupplierJob(env, { kind: 'submit-order', order_id: o.id }, 1)).toBe('submitted');
    const lines = (JSON.parse((await one<{ request_json: string }>('SELECT request_json FROM supplier_shipments WHERE order_id = ?', o.id)).request_json) as { lines: [string, number][] }).lines;
    const bad: string[] = [];
    for (const p of picks) {
      const c = card(p.source_path);
      const cj = cjOf(c);
      const opts = new Set((c.options?.skus ?? []).map((s) => s.id));
      const sold = (p.variants_full ?? []).find((v) => v.id === p.variants[0]!.id)!;
      const s = sold.supplier_sku;
      if (!lines.some(([l]) => l === s)) bad.push(`${p.id}: ${s} not in the request`);
      if (opts.size && !opts.has(s)) bad.push(`${p.id}: ${s} not an option`);
      if (cj?.product && s === cj.product) bad.push(`${p.id}: ${s} is the CJ product code`);
    }
    const lamp = lines.find(([s]) => s.startsWith('CJYD2247544'));
    console.log(`QA H2: ${picks.length} products (exclusions or one SKU) in one paid order; supplier lines ${lines.length}; floor lamp ${JSON.stringify(lamp)}; issues ${bad.length ? bad.join('; ') : 'none'}`);
    expect(lamp?.[0]).toBe('CJYD224754407GT');
    expect(bad).toEqual([]);
  });
});
