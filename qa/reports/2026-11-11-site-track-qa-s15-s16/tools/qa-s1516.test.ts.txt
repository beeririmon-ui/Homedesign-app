/** a supplier like a network one: creates the order the moment it receives it; can hang, fail, be asked, fail to answer */
function supplier(o: { hangMs?: number; failWith?: (i: SupplierOrderInput) => Error; find?: 'yes' | 'fails' | 'hangs'; cancel?: 'canceled' | 'requested' | 'refused' } = {}) {
  const orders: string[] = [];
  const cancels: string[] = [];
  const finds: string[] = [];
  const c: SupplierClient & { orders: string[]; cancels: string[]; finds: string[] } = {
    id: 'mock',
    orders,
    cancels,
    finds,
    async submitOrder(input) {
      if (o.failWith) throw o.failWith(input);
      orders.push(input.order_id);
      if (o.hangMs) await sleep(o.hangMs);
      return { supplier_order_id: `QA-SUP-${input.order_id}`, raw: {} };
    },
    async cancelOrder(sid) {
      cancels.push(sid);
      return { outcome: o.cancel ?? 'requested' };
    },
  };
  if (o.find)
    c.findOrder = async (our) => {
      finds.push(our);
      if (o.find === 'fails') throw Object.assign(new Error(`lookup 502 for consignee "${HE.full_name}"`), { status: 502 });
      if (o.find === 'hangs') await sleep(500);
      return orders.includes(our) ? { supplier_order_id: `QA-SUP-${our}` } : null;
    };
  return c;
}

/** a payment provider like a network one: counts calls and refunds; may dedupe, may be asked; a call may run past its deadline */
function payments(o: { idempotent?: boolean; find?: boolean; delayMs?: number; lateCompletes?: boolean } = {}) {
  const calls: string[] = [];
  const made: { key: string; amount: number }[] = [];
  const byKey = new Map<string, IssuedRefund>();
  const p: PaymentProvider & { calls: string[]; made: typeof made } = {
    id: 'mock',
    calls,
    made,
    refundsIdempotent: o.idempotent ?? true,
    async createSession() {
      throw new Error('not used');
    },
    async verifyWebhook() {
      return { ok: false, reason: 'not used' };
    },
    async refund(input: RefundInput, { idempotency_key }) {
      calls.push(idempotency_key);
      await sleep(o.delayMs ?? 15);
      if ((o.idempotent ?? true) && byKey.has(idempotency_key)) return byKey.get(idempotency_key)!;
      if (input.refunded_before_agorot + input.amount_agorot > input.paid_agorot) throw new RefundRefused('exceeds');
      if (o.lateCompletes === false) throw new Error('connection reset before the provider acted');
      made.push({ key: idempotency_key, amount: input.amount_agorot });
      const r = { refund_id: `QA-RF-${made.length}` };
      byKey.set(idempotency_key, r);
      return r;
    },
  };
  if (o.find) p.findRefund = async (k) => byKey.get(k) ?? null;
  return p;
}

    .replace(/\+/g, '-')
