async function cart(items: { variant_id: string; qty?: number }[]) {

/**
 * a supplier like a network one: creates the order the moment it receives it (a timed-out call still creates it); may
 * hang, fail, be asked (findOrder), dedupe by our order number (returns the first order), or neither
 */
function supplier(o: { hangMs?: number; failWith?: (i: SupplierOrderInput) => Error; failTimes?: number; find?: 'yes' | 'fails'; dedupes?: boolean } = {}) {
  const orders: string[] = [];
  const finds: string[] = [];
  let fails = o.failTimes ?? Infinity;
  const c: SupplierClient & { orders: string[]; finds: string[]; calls: number } = {
    id: 'mock',
    orders,
    finds,
    calls: 0,
    ...(o.dedupes ? { dedupesByOrderId: true } : {}),
    async submitOrder(input) {
      c.calls++;
      if (o.failWith && fails-- > 0) throw o.failWith(input);
      if (o.dedupes && orders.includes(input.order_id)) return { supplier_order_id: `QA-SUP-${input.order_id}-1`, raw: { deduplicated: true } };
      orders.push(input.order_id);
      const n = orders.length;
      if (o.hangMs) await sleep(o.hangMs);
      return { supplier_order_id: `QA-SUP-${input.order_id}-${n}`, raw: {} };
    },
  };
  if (o.find)
    c.findOrder = async (our) => {
      finds.push(our);
      if (o.find === 'fails') throw Object.assign(new Error(`lookup 502 for consignee ${LA.full_name}`), { status: 502 });
      const i = orders.indexOf(our);
      return i < 0 ? null : { supplier_order_id: `QA-SUP-${our}-${i + 1}` };
    };
  return c;
}

const brokenDb = (leak: string) =>
  new Proxy(env.DB, {
    get(t, k) {

      return (sql: string) => {
        if (/LEFT JOIN customers c ON c\.id = o\.customer_id WHERE o\.id/.test(sql)) return t.prepare(sql);
        throw new Error(leak);
      };
    },
  }) as unknown as D1Database;

describe('QA E (ת47): a send that died mid-call', () => {
  const health = async () => {
    const r = await req('/healthz');
    return `${r.status} ${((await r.json()) as { checks: { supplier: string } }).checks.supplier}`;
  };
  it('E1 my S15–S16 edge on the clock: a failure, then a retry that dies mid-call; and a first claim that died', async () => {
    vi.spyOn(console, 'error').mockImplementation(() => undefined);
    vi.spyOn(console, 'warn').mockImplementation(() => undefined);
    const T0 = Date.now();
    vi.useFakeTimers({ toFake: ['Date'], now: T0 });
    const e = await paid();
    // a 503 at T0 (failing_since T0, last_error set), then at T0+1 min the retry claims it and the isolate dies (no catch)
    await processSupplierJob(env, { kind: 'submit-order', order_id: e.id }, 1, { client: supplier({ failWith: () => Object.assign(new Error('503'), { status: 503 }) }) });
    vi.setSystemTime(T0 + 60_000);
    await env.DB.prepare("UPDATE supplier_shipments SET status = 'submitting', attempts = 2, updated_at = ? WHERE order_id = ?").bind(new Date(T0 + 60_000).toISOString(), e.id).run();
    const at: Record<string, string> = {};
    for (const m of [5, 10.5, 29.5, 30.5, 90]) {
      vi.setSystemTime(T0 + m * 60_000);
      at[`T0+${m}m`] = await health();
    }
    // a first claim that died (no failure before it): claimed at T1, no failing_since, no last_error
    await env.DB.prepare("UPDATE orders SET status = 'canceled' WHERE id = ?").bind(e.id).run();
    const T1 = T0 + 2 * 3600_000;
    vi.setSystemTime(T1);
    const f = await paid();
    await env.DB.prepare("UPDATE supplier_shipments SET status = 'submitting', attempts = 1, updated_at = ? WHERE order_id = ?").bind(new Date(T1).toISOString(), f.id).run();
    const first: Record<string, string> = {};
    for (const m of [11, 29.5, 30.5]) {
      vi.setSystemTime(T1 + m * 60_000);
      first[`T1+${m}m`] = await health();
    }
    // and the takeover: the next delivery after the lease asks (it can't be asked here → a person); /healthz then
    vi.setSystemTime(T1 + 31 * 60_000);
    const r = await processSupplierJob(env, { kind: 'submit-order', order_id: f.id }, 2, { client: supplier() });
    const s = await ship(f.id);
    first['takeover'] = `${r} → shipment ${s.status}/${s.reach}, order ${await status(f.id)}, /healthz ${await health()}`;
    vi.useRealTimers();
    console.log(`QA E1: retry died mid-call ${JSON.stringify(at)} | first claim died ${JSON.stringify(first)} | lease ${SUBMIT_LEASE_MS / 60000} min`);
    expect(at['T0+5m']!.startsWith('200')).toBe(true);
    expect(at['T0+30.5m']).toBe('503 degraded');
    expect(at['T0+90m']).toBe('503 degraded');
    expect(first['T1+29.5m']!.startsWith('200')).toBe(true);
    expect(first['T1+30.5m']).toBe('503 degraded');
  });
});

    const res: Record<string, string> = {};
    // the timed-out call created nothing (failWith once): asked → none → sent once
    const none = supplier({ failWith: () => Object.assign(new Error('socket hang up'), { status: 503 }), failTimes: 1, find: 'yes' });
    const o1 = await paid();
    const outs1 = [await processSupplierJob(env, { kind: 'submit-order', order_id: o1.id }, 1, { client: none })];
    outs1.push(await processSupplierJob(env, { kind: 'submit-order', order_id: o1.id }, 2, { client: none }));
    res.none = `${outs1.join(',')}; supplier orders ${none.orders.length}, finds ${none.finds.length}; order ${await status(o1.id)}`;
    // the lookup fails every time: never sent again; supplier_error at attempt 5; a person's check is then possible
    const fails = supplier({ hangMs: 40, find: 'fails' });
    const o2 = await paid();
    const outs2 = [await processSupplierJob(env, { kind: 'submit-order', order_id: o2.id }, 1, { client: fails, callTimeoutMs: 15 })];
    await sleep(60);
    for (let a = 2; a <= 6; a++) outs2.push(await processSupplierJob(env, { kind: 'submit-order', order_id: o2.id }, a, { client: fails }));
    const s2 = await ship(o2.id);
    res.lookup_fails = `${outs2.join(',')}; supplier orders ${fails.orders.length}, calls ${fails.calls}, finds ${fails.finds.length}; shipment ${s2.status}/${s2.reach}, last_error "${s2.last_error}"; order ${await status(o2.id)}`;
    const chk = await req(`/api/admin/orders/${o2.id}/supplier-check`, json({ outcome: 'found', supplier_order_id: `QA-SUP-${o2.id}-1`, note: 'נמצאה אצל הספק' }, 'POST', admin));
    res.lookup_fails_person_found = `${chk.status}; order ${await status(o2.id)}; shipment ${(await ship(o2.id)).status}; supplier orders ${fails.orders.length}`;
    // a refusal after a maybe: the refusal created nothing, the reach stays as it was
    const ref = supplier({ hangMs: 40, find: 'yes' });
    const o3 = await paid();
    await processSupplierJob(env, { kind: 'submit-order', order_id: o3.id }, 1, { client: ref, callTimeoutMs: 15 });
    await sleep(60);
    ref.orders.length = 0; // the supplier lost it (it never really took it): asked → none → the send is refused
    const refusing: SupplierClient = { ...ref, id: 'mock', findOrder: ref.findOrder, submitOrder: async () => { throw new SupplierRefused('sku_out_of_stock'); } };
    const r3a = await processSupplierJob(env, { kind: 'submit-order', order_id: o3.id }, 2, { client: refusing });
    const s3a = await ship(o3.id);
    const r3b = await processSupplierJob(env, { kind: 'submit-order', order_id: o3.id }, 3, { client: ref });
    res.refusal_after_maybe = `${r3a} (${s3a.status}/${s3a.reach}) → ${r3b} (${(await ship(o3.id)).status}/${(await ship(o3.id)).reach}); supplier orders ${ref.orders.length}, finds ${ref.finds.length}`;
    process.stdout.write(`QA B2: ${JSON.stringify(res)}\n`);
    expect(none.orders).toHaveLength(1);
    expect(fails.orders).toHaveLength(1);
  });

  it('B3 the one left to a person, through the queue consumer and the admin API: "none" sends once; a second "none" is refused; "found" records', async () => {
    quiet();
    const res: Record<string, string> = {};
    for (const outcome of ['none', 'found'] as const) {
      const sup = supplier({ hangMs: 40 });
      const o = await paid();
      await processSupplierJob(env, { kind: 'submit-order', order_id: o.id }, 1, { client: sup, callTimeoutMs: 15 });
      await sleep(60);
      // the queue's own retry, through src/index.ts with the configured (mock) supplier: the mock can't be asked
      const acks: string[] = [];
      await worker.queue!(
        {
          queue: 'q',
          messages: [{ id: 'm', timestamp: new Date(), body: { kind: 'submit-order', order_id: o.id }, attempts: 2, ack: () => void acks.push('ack'), retry: () => void acks.push('retry') }],
          ackAll() {},
          retryAll() {},
        } as unknown as MessageBatch<SupplierJob>,
        env,
      );
      const mid = `${await status(o.id)} ${(await ship(o.id)).status}/${(await ship(o.id)).reach} [${acks.join(',')}]`;
      const body = outcome === 'found' ? { outcome, supplier_order_id: `QA-SUP-${o.id}-1`, note: 'נמצאה' } : { outcome, note: 'אין אצל הספק' };
      const r1 = await req(`/api/admin/orders/${o.id}/supplier-check`, json(body, 'POST', admin));
      const queued = sent.filter((j) => j.order_id === o.id).length;
      const r2 = await req(`/api/admin/orders/${o.id}/supplier-check`, json(body, 'POST', admin));
      const queued2 = sent.filter((j) => j.order_id === o.id).length;
      let after = '';
      if (outcome === 'none') {
        // the queued send runs (the timed-out call really created nothing in this story: the supplier forgets it)
        sup.orders.length = 0;
        after = await processSupplierJob(env, { kind: 'submit-order', order_id: o.id }, 1, { client: sup });
      }
      res[outcome] = `after the queue's retry: ${mid}; check ${r1.status}, sends queued ${queued}; second check ${r2.status} (${((await r2.json()) as { error?: string }).error ?? 'ok'}), queued ${queued2}; then ${after || '-'}: order ${await status(o.id)}, shipment ${(await ship(o.id)).status}, supplier orders ${sup.orders.length}`;
    }
    process.stdout.write(`QA B3: ${JSON.stringify(res)}\n`);
    expect(res.none).toContain('sends queued 1');
    expect(res.none).toContain('queued 1;');
  });
